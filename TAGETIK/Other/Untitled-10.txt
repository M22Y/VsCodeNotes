SELECT * FROM (
SELECT 
  COD_SCENARIO AS SCENARIO,
  COD_PERIODO AS PERIOD,
  COD_AZIENDA AS ENTITY, 
  COD_CONTO AS ACCOUNT, 
  COD_DEST2 AS COST_CENTER, 
  COD_DEST5 AS SUPPLIER, 
  COD_DEST1 AS SNA,
  COD_DEST3 AS PROJECT,
  COD_VALUTA AS CURRENCY,
  SUM(IMPORTO) BUDGET_PY
FROM  V_DATI_SALDI_FULL_MENSILI
WHERE  COD_PERIODO >= 3
AND   COD_SCENARIO = '2026BUD_P'
AND   COD_CATEGORIA IN (
        SELECT COD_CATEGORIA
        FROM categoria_gerarchia_abbim
      WHERE  (
        COD_CATEGORIA_ELEGER ='PLAN_FW'
        )
      OR (  
        COD_CATEGORIA_ELEGER LIKE 'GR_P%'
        AND RIGHT(COD_CATEGORIA_ELEGER,2) < 3
        )
      )
GROUP BY
  COD_SCENARIO,
  COD_AZIENDA, 
  COD_PERIODO,
  COD_CONTO, 
  COD_DEST2,
  COD_DEST5, 
  COD_DEST1, 
  COD_DEST3, 
  COD_VALUTA
)  AS S

WHERE  S.ENTITY ='601'
AND S.COST_CENTER = '8250'
AND S.ACCOUNT= '54102'
AND S.SUPPLIER = 'F001349'
ORDER BY S.PERIOD
