DECLARE @DELETE_SQL_ACT NVARCHAR(4000)
DECLARE @DELETE_SQL_FCT NVARCHAR(4000)
DECLARE @INSERT_SQL_FCT NVARCHAR(4000)
DECLARE @INSERT_SQL_ACT  NVARCHAR(4000)
DECLARE @DEBUG NVARCHAR(4000)
DECLARE @ALL_COLUMNS VARCHAR(4000)
DECLARE @SOURCESCN VARCHAR(4000)
DECLARE @ACTUALSCN VARCHAR(4000)
DECLARE @ACTLMONTH VARCHAR(4000)
DECLARE @TARGETSCN VARCHAR(4000)
DECLARE @TARGETLCK VARCHAR(4000)
DECLARE @MONTHLCK VARCHAR(4000)
DECLARE @DEBUG_SQL VARCHAR(4000)
DECLARE @TARGET_SQL VARCHAR(4000)
DECLARE @SCNGROUP VARCHAR(4000)

BEGIN
/*Select source month and closing month for actuals in parameters via scenario attributes*/
--SET @SOURCESCN = ${$Scenario.&SOURCE_SCENARIO&.code}
--SET @ACTUALSCN  =  ${$Scenario.&ACTUAL_SCENARIO&.code}

SET @SOURCE_TABLE_NAME =  ${DATASETPARTITION("002", "CYT001", PROCESS($Scenario.code, $Period.code).code, $Scenario.code, $Period.code,,,"")}

SET @SCENARIO =  ${$Scenario.code}
SET @PERIOD =  ${$Period.code}
SET @TARGET_SCENARIO =  (SELECT DISTINCT )
SET @TARGET_PERIOD =  ${$Period.code}
-- SET @SCNGROUP =  '&SCENARIO_GROUPING&'
-- FROM ${DATASETPARTITION("002", "000001", PROCESS($Scenario.attribute5.code, ).code, $Scenario.attribute5.code, "01",,,"(SELECT NULL AS SCENARIO, NULL AS PERIOD,NULL AS ACCOUNT, NULL AS ENTITY, NULL AS CATEGORY, NULL AS DIVISION, NULL AS CUSTOMER, NULL AS PRODUCT, NULL AS RSRCE, NULL AS OPE_TYPE, NULL AS TEXT_ORIGIN, 0 AS AMOUNT)")} AS T1
-- FROM ${DATASETPARTITION("002", "CYT001", PROCESS($Scenario.code, $Period.code).code, $Scenario.attribute5.code, "01",,,"(SELECT NULL AS SCENARIO, NULL AS PERIOD,NULL AS ACCOUNT, NULL AS ENTITY, NULL AS CATEGORY, NULL AS DIVISION, NULL AS CUSTOMER, NULL AS PRODUCT, NULL AS RSRCE, NULL AS OPE_TYPE, NULL AS TEXT_ORIGIN, 0 AS AMOUNT)")} AS T1

-- FILTERS	
--PROVENIENZA <> 'MAP_100_01'
--AND TGK_COD_SCENARIO_ETL = ${$SCENARIO.CODE}
--AND TGK_COD_PERIODO_ETL = ${$PERIOD.CODE}
--AND TGK_COD_AZIENDA IN (SELECT COD_AZIENDA FROM AZIENDA_GERARCHIA_ABBI WHERE COD_AZIENDA_GERARCHIA='800' AND COD_AZIENDA_ELEGER='CYR')"


/*Target is automatically selected based on the process scenario it is running from. Passed as #Scenarios# in parameter*/
-- SET @TARGETSCN = '&TARGET_SCENARIO&'



/*Generate the delete statement to delete ALL rows for the target scenario*/
SET @DELETE_SQL_TB =
 'DELETE FROM &TABLE_NAME&
      WHERE COD_SCENARIO IN (SELECT GRP.COD_SCENARIO FROM SCENARIO_GERARCHIA_ABBIM AS GRP WHERE GRP.COD_SCENARIO_ELEGER = ''' + @SCNGROUP  + ''')
	    AND COD_PERIODO  <= ''' + @ACTLMONTH  + ''' ' 



/*Create list of columns for the source and target table*/
SELECT @ALL_COLUMNS = COALESCE(@ALL_COLUMNS + ', ', '') + column_Name
FROM INFORMATION_SCHEMA.COLUMNS where table_NAME = '&TABLE_NAME&' AND column_name NOT IN ('OID_&TABLE_NAME&', 'COD_SCENARIO', 'PROVENIENZA', 'USERUPD', 'DATEUPD')

/* Generate ACT Insert Statement */
SET @INSERT_SQL_ACT = 
'
WITH 
GRP_SCN AS
(
SELECT 
GRP.COD_SCENARIO AS TARGET_SCN, 
(SELECT SCN.attributo2 FROM SCENARIO AS SCN WHERE SCN.COD_SCENARIO = GRP.COD_SCENARIO) AS SOURCE_SCN 
FROM SCENARIO_GERARCHIA_ABBIM AS GRP 
WHERE GRP.COD_SCENARIO_ELEGER = ''' + @SCNGROUP  + '''
),
DATI AS
(
Select * from &TABLE_NAME& DSL 
)
INSERT INTO &TABLE_NAME&
(
    ' + @ALL_COLUMNS + ',
    COD_SCENARIO,
    OID_&TABLE_NAME&,
    PROVENIENZA,
    USERUPD,
    DATEUPD
)
Select ' + @ALL_COLUMNS + ',
       GRP_SCN.TARGET_SCN,
	   NEWID() AS OID,
       PROVENIENZA,
    ''#USER#'' AS USERUPD,
       CURRENT_TIMESTAMP AS DATEUPD
from DATI 
INNER JOIN GRP_SCN ON DATI.COD_SCENARIO = GRP_SCN.SOURCE_SCN
AND DATI.COD_PERIODO <=  ''' + @ACTLMONTH  + ''' ' 


/* Generate ACT Insert Statement */
SET @INSERT_SQL_TB = 
'
WITH
PERIOD_FILTER AS (
SELECT COD_PERIODO
FROM SCENARIO_PERIODO
WHERE COD_RACCOLTA='#PROCESS#'
AND FLAG_BLOCCATO = 0
AND COD_SCENARIO=(SELECT COD_SCENARIO FROM SCENARIO_PERIODO WHERE COD_RACCOLTA ='#PROCESS#' ORDER BY COD_SCENARIO LIMIT 1)
),
PLACEHOLDER AS (
SELECT NULL AS SCENARIO, NULL AS PERIOD, NULL AS ENTITY, NULL AS ACCOUNT, NULL AS COST_CENTER, NULL AS SNA, NULL AS PROJECT, NULL AS SUPPLIER, NULL AS CATEGORY, NULL AS CURRENCY, NULL AS AMOUNT
FROM DUMMY
),
DATAPULL AS (
SELECT 
     SCENARIO, PERIOD, ENTITY, ACCOUNT, COST_CENTER, SNA, PROJECT, SUPPLIER, CATEGORY, CURRENCY, AMOUNT
FROM &TARGET_TABLE_NAME& PLACEHOLDER
'



/*Delete Actual month rows and insert actuals*/
Exec (@DELETE_SQL_TB )
Exec (@INSERT_SQL_TB )

END
