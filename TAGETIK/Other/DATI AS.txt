DATI AS
(
    SELECT ' + @ALL_COLUMNS + ', COD_SCENARIO, PROVENIENZA
    FROM &TABLE_NAME& 
    WHERE COD_PERIODO <= ''' + @ACTLMONTH  + '''
)
INSERT INTO &TABLE_NAME&
(
    ' + @ALL_COLUMNS + ',
    COD_SCENARIO,
    OID_&TABLE_NAME&,
    PROVENIENZA,
    USERUPD,
    DATEUPD
)
SELECT 
    ' + @ALL_COLUMNS + ',
    GRP_SCN.TARGET_SCN,
    NEWID() AS OID,
    GRP_SCN.SOURCE_SCN + ''' + @ACTLMONTH + ''' AS PROVENIENZA,
    ''#USER#'' AS USERUPD,
    CURRENT_TIMESTAMP AS DATEUPD
FROM DATI
INNER JOIN GRP_SCN ON DATI.COD_SCENARIO = GRP_SCN.SOURCE_SCN
'
