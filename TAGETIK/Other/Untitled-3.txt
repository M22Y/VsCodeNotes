
-- Updated IMPORTO to IMPORTO2, to reflect the changes in the source table
SELECT 
  COD_AZIENDA AS ENTITY, 
  COD_CONTO AS ACCOUNT, 
  COD_DEST2 AS COST_CENTERS, 
  COD_DEST5 AS FOURNISSEUR, 
  COD_DEST3 AS PROJECT,
  COD_DEST1 AS SNA,
  COD_VALUTA2 AS CURRENCY,
  SUM(IMPORTO2) TOT,
  COD_AZIENDA || '_' || COD_CONTO || '_' || COD_DEST2 || '_' || COD_DEST5 || '_' || COD_DEST1 || '_' || COD_DEST3 || '_' || COD_VALUTA2 AS CONCAT_COLUMN
  
  
FROM  V_DATI_SALDI
WHERE COD_PERIODO ='12'
AND   COD_SCENARIO = '2026BUD_P'
AND   IMPORTO2<>0
AND   COD_CATEGORIA IN (
        SELECT COD_CATEGORIA
        FROM categoria_gerarchia_abbim
      WHERE  (
        COD_CATEGORIA_ELEGER ='PLAN_FW'
        )
      OR (  
        COD_CATEGORIA_ELEGER LIKE 'GR_P%'
        AND RIGHT(COD_CATEGORIA_ELEGER,2) < '03'
        )
      )
GROUP BY COD_AZIENDA || '_' || COD_CONTO || '_' || COD_DEST2 || '_' || COD_DEST5 || '_' || COD_DEST1 || '_' || COD_DEST3 || '_' || COD_VALUTA2,
  COD_AZIENDA, 
  COD_CONTO, 
  COD_DEST2,
  COD_DEST5, 
  COD_DEST1, 
  COD_DEST3, 
  COD_VALUTA2
ORDER BY CONCAT_COLUMN
