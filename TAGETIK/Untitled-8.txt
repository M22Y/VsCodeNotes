WITH /*Filter on non locked period*/
PERIOD_FILTER AS (
  SELECT COD_PERIODO
  FROM SCENARIO_PERIODO
  WHERE COD_RACCOLTA='#PROCESS#'
  AND FLAG_BLOCCATO = 0
  AND COD_SCENARIO=(SELECT COD_SCENARIO FROM SCENARIO_PERIODO WHERE COD_RACCOLTA ='#PROCESS#' ORDER BY COD_SCENARIO LIMIT 1)
),

DATAPULL AS ( /* Collect the data from FULL_MENSILI, for all the available periods*/
SELECT 
  COD_SCENARIO SCENARIO,COD_PERIODO PERIOD, COD_AZIENDA ENTITY, COD_CONTO ACCOUNT, COD_DEST2 COST_CENTER, COD_DEST1 SNA, COD_DEST3 PROJECT, COD_DEST5 SUPPLIER, COD_CATEGORIA CATEGORY, COD_VALUTA2 CURRENCY, IMPORTO2 AMOUNT
FROM V_DATI_SALDI_FULL_MENSILI 
WHERE 
COD_CATEGORIA IN (
    SELECT COD_CATEGORIA
    FROM categoria_gerarchia_abbim
   WHERE (
    COD_CATEGORIA_ELEGER ='PLAN_FW'
    )
   OR ( 
    COD_CATEGORIA_ELEGER LIKE 'GR_P%'
    AND RIGHT(COD_CATEGORIA_ELEGER,2) <= (SELECT DISTINCT ATTRIBUTO1 FROM SCENARIO WHERE COD_SCENARIO=@P0010@)
    )
   ) /* Filter on Category for the Transfert in case of budget Scenario and all PLAN_FW*/
AND COD_SCENARIO =@P0010@ /* filter on Historical scenario*/
),


MONTHLY_VIEW AS (
  SELECT 
    SCENARIO, RIGHT('0' || CAST(PERIOD AS VARCHAR), 2) PERIOD, ENTITY, ACCOUNT, COST_CENTER, SNA, PROJECT, SUPPLIER,  CURRENCY, SUM(AMOUNT) AMOUNT
  FROM DATAPULL

  WHERE 1=1
  AND PERIOD IN (SELECT COD_PERIODO FROM PERIOD_FILTER)
  GROUP BY 
    SCENARIO, 
   PERIOD,
    ENTITY, ACCOUNT, COST_CENTER, SNA, PROJECT, SUPPLIER,  CURRENCY 
  ) 

SELECT  monthly.SCENARIO,
    monthly.PERIOD,
    monthly.ENTITY,
    monthly.ACCOUNT,
    monthly.COST_CENTER,
    monthly.SNA, 
    monthly.PROJECT,
    monthly.SUPPLIER,
    monthly.CURRENCY,
    Sum(monthly.m_amount/ tot.m_tot) AS perc_driver
FROM   (
         SELECT  SCENARIO,
             PERIOD,
             ENTITY,
             ACCOUNT,
             COST_CENTER,
             SNA,
             PROJECT,
             SUPPLIER,
             CURRENCY,
             Sum(AMOUNT) AS m_amount
         FROM   MONTHLY_VIEW
         WHERE  SCENARIO=@P0010@
         AND   PERIOD IN (SELECT PERIOD FROM PERIOD_FILTER)
         GROUP BY SCENARIO,
             PERIOD,
             ENTITY,
             ACCOUNT,
             COST_CENTER,
             SNA,
             PROJECT,
             SUPPLIER,
             CURRENCY
         HAVING  sum(AMOUNT)<>0 ) monthly
JOIN
    (
         SELECT  SCENARIO,
             ENTITY,
             ACCOUNT,
             COST_CENTER,
             SNA,
             PROJECT,
             SUPPLIER,
             CURRENCY,
             sum(AMOUNT) AS m_tot
         FROM   MONTHLY_VIEW
         WHERE  SCENARIO=@P0010@
         AND   PERIOD IN (SELECT PERIOD FROM PERIOD_FILTER)
         GROUP BY SCENARIO,
             ENTITY,
             ACCOUNT,
             COST_CENTER,
             SNA,
             PROJECT,
             SUPPLIER,
             CURRENCY
         HAVING  sum(AMOUNT)<>0 ) tot
ON MONTHLY.SCENARIO=TOT.SCENARIO
  AND MONTHLY.ENTITY=TOT.ENTITY
  AND MONTHLY.ACCOUNT=TOT.ACCOUNT
  AND MONTHLY.COST_CENTER=TOT.COST_CENTER
  AND MONTHLY.SNA=TOT.SNA
  AND MONTHLY.PROJECT=TOT.PROJECT
  AND MONTHLY.SUPPLIER=TOT.SUPPLIER

  AND MONTHLY.CURRENCY=TOT.CURRENCY
GROUP BY monthly.SCENARIO,
    monthly.PERIOD,
    monthly.ENTITY,
    monthly.ACCOUNT,
    monthly.COST_CENTER,
    monthly.SNA,
    monthly.PROJECT,
    monthly.SUPPLIER,
    monthly.CURRENCY
