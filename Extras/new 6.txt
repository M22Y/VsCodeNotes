WHEN ${$ANL_CO_00007.desc} IN (SELECT DESC1 FROM ED_CO_00007 WHERE COD= ${$ANL_CO_00007.code}) AND ${Period.offset(-11).code} like '01' THEN /*FR*/ '1' + NCHAR(0x1D49) +  NCHAR(0x02B3)  + ' ' + F_TGK_UNICODE('decode', DESC_PERIODO1)
