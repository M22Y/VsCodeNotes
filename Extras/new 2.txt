CASE
WHEN ${user.reportingLang.code}='en' THEN NOTE_LABEL_EN
WHEN ${user.reportingLang.code}='fr'  THEN NOTE_LABEL_FR
CASE WHEN ${$Period.desc} =DESC_PERIODO0 THEN NOTE_LABEL_EN
WHEN ${$Period.desc} = DESC_PERIODO2 THEN NOTE_LABLEL_FR 

ELSE NOTE_LABEL_EN
END AS NOTE_LABEL
FROM 
${DATASETPARTITION("003", "TE0003",process($Scenario.code,$Period.code).code, $Scenario.code, $Period.code)}

WHERE 

NOTE_ID IN('INVES_CV_FV',
'INVES_BNDLOAN',
'INVES_INV_RES',
'INVES_INV_INC')
AND SCEN_TYPE =${$ANL_CO_00007.code}
