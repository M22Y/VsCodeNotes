/* Prepopulation based on Balance type Dimension */
/* PCC Scenario Type */
SELECT DISTINCT
    ${$Scenario.code} AS SCENARIO,
    ${$Period.code} AS PERIOD,
    'CONS_PCC_CAD' AS SCEN_TYPE,
    B.COD AS NOTE_ID,
    /* CONTO.COD_CONTO AS ACC_COD, */
	BAL_TYPE.COD AS BAL_TYPE_COD,
    BAL_TYPE.DESC0 AS BAL_TYPE_DESC_EN,
    BAL_TYPE.DESC1 AS BAL_TYPE_DESC_FR
FROM ED_CO_00002 BAL_TYPE
JOIN ED_CO_00006 AS B ON (BAL_TYPE.COD LIKE '%' || B.COD || '%')

UNION ALL

/* PFC */
SELECT DISTINCT
    ${$Scenario.code} AS SCENARIO,
    ${$Period.code} AS PERIOD,
    'CONS_PFC_CAD' AS SCEN_TYPE,
    B.COD AS NOTE_ID,
    /* CONTO.COD_CONTO AS ACC_COD, */
	BAL_TYPE.COD AS BAL_TYPE_COD,
    BAL_TYPE.DESC0 AS BAL_TYPE_DESC_EN,
    BAL_TYPE.DESC1 AS BAL_TYPE_DESC_FR
FROM ED_CO_00002 BAL_TYPE
JOIN ED_CO_00006 AS B ON (BAL_TYPE.COD LIKE '%' || B.COD || '%')

 SELECT

    FROM ${DATASETPARTITION("003", "TE0004", "ID", $Scenario.code, $Period.offset(-3).code)}
